{{- /* Hook renderowania odnośników w treści (Markdown). Dzięki niemu w tekście wystarczy pisać ścieżki polskie, np.: [statystyki](/statystyki/) [wzór ponaglenia](/wzory/ponaglenie-bezczynnosc/) Hugo sam zamienia je na właściwy adres w danym języku (/en/statistics/, /ru/obraztsy/…) i dokłada przedrostek baseURL, gdy strona stoi w podkatalogu (np. /tuzyjemy/). Jeśli strona o takiej ścieżce nie istnieje w bieżącym języku, odnośnik prowadzi do istniejącej wersji w innym języku. Dopiero gdy strona nie istnieje nigdzie, adres jest używany dosłownie z poprawnym przedrostkiem języka. Przy odnośnikach zewnętrznych prowadzących do strony w innym języku niż czytany (wg data/linklangs.yaml) dokładamy obok mały odnośnik do Tłumacza Google. */ -}} {{- $d := .Destination -}} {{- $external := or (hasPrefix $d "http://") (hasPrefix $d "https://") (hasPrefix $d "mailto:") (hasPrefix $d "tel:") (hasPrefix $d "//") -}} {{- $href := $d -}} {{- if and (not $external) (hasPrefix $d "/") -}} {{- $parts := split $d "#" -}} {{- $path := index $parts 0 -}} {{- $frag := "" -}} {{- if gt (len $parts) 1 -}}{{- $frag = printf "#%s" (index $parts 1) -}}{{- end -}} {{- $lookup := strings.TrimSuffix "/" $path -}} {{- with site.GetPage $lookup -}} {{- $href = printf "%s%s" .RelPermalink $frag -}} {{- else -}} {{- $found := false -}} {{- range hugo.Sites -}} {{- if not $found -}} {{- with .GetPage $lookup -}} {{- $href = printf "%s%s" .RelPermalink $frag -}} {{- $found = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- $href = printf "%s%s" ($path | relLangURL) $frag -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Język strony docelowej. Szukamy po kolei: najpierw pełny adres, potem nazwa hosta (bez „www."), na końcu domena nadrzędna. Pełny adres jest pierwszy, bo na niektórych serwisach język zależy od konkretnej strony, a nie od hosta — tak jest z naszymi dokumentami w Google Docs, gdzie pod tym samym hostem leżą pliki w różnych językach. */ -}} {{- $tl := .Page.Site.Language.Lang -}} {{- $sl := "" -}} {{- $host := "" -}} {{- if and $external (hasPrefix $d "http") -}} {{- $host = strings.TrimPrefix "www." (urls.Parse $d).Host -}} {{- with index hugo.Data.linklangs $d -}} {{- $sl = . -}} {{- else -}} {{- with index hugo.Data.linklangs $host -}} {{- $sl = . -}} {{- else -}} {{- /* dopasowanie po domenie nadrzędnej, np. migrant.wsc.gov.pl → gov.pl */ -}} {{- range $dom, $lang := hugo.Data.linklangs -}} {{- if and (not $sl) (hasSuffix $host (printf ".%s" $dom)) -}}{{- $sl = $lang -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Go czyści adresy w szablonach i podmienia nieznane schematy na „#ZgotmplZ". Schemat „tel:" nie jest na jego liście, więc numer telefonu przestawał być odnośnikiem — dlatego tylko dla „tel:" oznaczamy adres jako bezpieczny. */ -}} {{ .Text | safeHTML }} {{- /* Przy części adresów tłumacz nie ma czego pokazać (patrz data/linknotranslate.yaml) — zostawiamy wtedy samą plakietkę języka, żeby odnośnik nie prowadził do pustej strony. */ -}} {{- $translatable := true -}} {{- with hugo.Data.linknotranslate -}} {{- if in . $host -}}{{- $translatable = false -}}{{- end -}} {{- end -}} {{- if and $sl (ne $sl $tl) -}} {{ upper $sl }}{{ if $translatable }}{{ $u := printf "https://translate.google.com/translate?sl=%s&tl=%s&u=%s" $sl $tl ($d | urlquery) }}🌐 {{ i18n "translate_page" }}{{ end }} {{- end -}}
