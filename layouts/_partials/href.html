{{- /* Zamienia ścieżkę z front matter na poprawny adres. Wywołanie: {{ partial "href.html" "/statystyki/" }} Działa tak samo jak hook odnośników w treści: jeśli istnieje strona o takiej ścieżce, zwraca jej adres w bieżącym języku; jeśli nie — dokłada tylko przedrostek języka i baseURL. */ -}} {{- $d := . -}} {{- $out := $d -}} {{- if or (hasPrefix $d "http://") (hasPrefix $d "https://") (hasPrefix $d "mailto:") -}} {{- $out = $d -}} {{- else if hasPrefix $d "/" -}} {{- $lookup := strings.TrimSuffix "/" $d -}} {{- with site.GetPage $lookup -}} {{- $out = .RelPermalink -}} {{- else -}} {{- /* Strony nie ma w bieżącym języku. Zamiast wskazywać adres, który da 404, szukamy jej w pozostałych wersjach językowych — tak samo jak listy dokumentów pokazują wersję zastępczą. */ -}} {{- $found := false -}} {{- range hugo.Sites -}} {{- if not $found -}} {{- with .GetPage $lookup -}} {{- $out = .RelPermalink -}} {{- $found = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- $out = $d | relLangURL -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
